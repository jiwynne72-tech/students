sw.js

Save this file as sw.js in your repository root directory to enable offline PWA caching.

const CACHE_NAME = 'english-portal-v1';
const ASSETS_TO_CACHE = [
  './',
  './index.html',
  './manifest.json',
  './logo.png',
  './bg.png',
  './l1.html'
];

self.addEventListener('install', (event) => {
  event.waitUntil(
    caches.open(CACHE_NAME).then((cache) => {
      return cache.addAll(ASSETS_TO_CACHE);
    })
  );
  self.skipWaiting();
});

self.addEventListener('activate', (event) => {
  event.waitUntil(
    caches.keys().then((cacheNames) => {
      return Promise.all(
        cacheNames.map((cache) => {
          if (cache !== CACHE_NAME) {
            return caches.delete(cache);
          }
        })
      );
    })
  );
  self.clients.claim();
});

self.addEventListener('fetch', (event) => {
  event.respondWith(
    caches.match(event.request).then((response) => {
      return response || fetch(event.request);
    })
  );
});

Student Repository Duplication Workflow

Why PNG Logo? Standard JPG images contain solid white background pixels that form a square box around circular emblems. A transparent PNG (`logo.png`) contains an alpha channel so only the circular graphic renders cleanly over dark or animated backgrounds.

Step 1: Create a template repository on GitHub named english-portal-template containing index.html, logo.png, bg.png, manifest.json, and sw.js.

Step 2: Duplicate repository for a student:
git clone https://github.com/yourusername/english-portal-template.git english-giorgia

Step 3: In index.html, replace Giorgia with the student's name in the header element.

Step 4: Update lesson cards & link to new pages (l1.html, l2.html) after each session.
